   string EventSide(const string level_key,const string comment) const
     {
      string side_key=(level_key!="" ? level_key : comment);

         side_key=="STR ORB" ||
         side_key=="STR AVB")
         return "buy";
      if(StringFind(side_key,"STR S")==0 ||
         side_key=="STR ORS" ||
         side_key=="STR AVS")
         return "sell";
      return "";
     }
